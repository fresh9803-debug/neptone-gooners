from flask import Flask, request import logging import requests app = Flask(__name__) logging.basicConfig(filename='ip_log.txt', level=logging.INFO) def get_address(ip): try: response = requests.get(f'http://ip-api.com/json/{ip}').json() if 'lat' in response: return f"{response['city']}, {response['region']}, {response['country']}" else: return "Unknown" except Exception as e: logging.warning(f"Failed to get address for {ip}: {e}") return "Unknown" @app.route('/') def log_ip(): ip = request.remote_addr address = get_address(ip) logging.info(f'IP Logged: {ip} - Address: {address}') return 'Hello!' if __name__ == '__main__': app.run(debug=True)
